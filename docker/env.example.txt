# Mini RAG Chat Docker 环境变量配置
# 将文件重命名为.env，默认在 docker 目录下，docker-compose.yml 会自动读取

# ==================== 基础配置 ====================
# 基础数据目录（所有数据将存储在此目录下）
BASE_DIR=/opt/mini-rag

# 网络配置
NETWORK_NAME=rag-network

# ==================== Mini RAG Chat 服务配置 ====================
# 镜像名称
MINI_RAG_IMAGE=mini-rag-chat:latest

# 容器名称
MINI_RAG_CONTAINER_NAME=mini-rag-chat

# 端口配置（格式：绑定地址:容器端口）
MINI_RAG_PORT=127.0.0.1:5000:5000

# Flask 环境
FLASK_ENV=production

# 配置文件路径（容器内路径）
CONFIG_PATH=/app/config.yaml

# ==================== Ollama 服务配置 ====================
# 容器名称
OLLAMA_CONTAINER_NAME=mini-rag-ollama

# Ollama 端口（仅内部网络，不暴露到公网）
OLLAMA_PORT=11434

# Ollama 访问来源限制
OLLAMA_ORIGINS=http://localhost,http://mini-rag,http://nginx
# OLLAMA_ORIGINS=*  # 危险：允许所有来源（取消注释以启用）

# ==================== 目录路径配置 ====================
# 代码目录（项目源代码所在目录，代码和数据都在BASE_DIR下）
CODE_DIR=${BASE_DIR}

# Ollama 数据目录
OLLAMA_DATA_DIR=${BASE_DIR}/ollama/data

# Ollama 模型目录
OLLAMA_MODEL_DIR=${BASE_DIR}/ollama/model

# Mini RAG 配置文件
CONFIG_FILE=${BASE_DIR}/config.yaml

# 知识库数据目录
DATA_DIR=${BASE_DIR}/data

# 新数据目录
DATA_NEW_DIR=${BASE_DIR}/data_new

# 向量存储目录
VECTOR_STORE_DIR=${BASE_DIR}/vector_store

# 模型目录
MODEL_DIR=${BASE_DIR}/model

# 前端模板目录
TEMPLATES_DIR=${BASE_DIR}/templates

# 日志目录
LOG_DIR=${BASE_DIR}/log

# ==================== 服务间通信配置 ====================
# Ollama 服务地址（容器内访问）
OLLAMA_BASE_URL=http://ollama:11434
